window.__ModuleLoader__.load({
  id: 'qiyao-calendar',
  factory(require) {
    const React = require('react')
    const h = React.createElement
    const DATA = /*__DATA__*/ null

    // 日期与事项解析只有一份实现：shared/items.mjs，构建时原样内联成 QIYAO
    /*__SHARED__*/

    const POPOVER_ID = 'qiyao-calendar-popover'
    let PLUGIN_CTX = null

    const chipStyle = {
      display: 'inline-block',
      padding: '1px 8px',
      borderRadius: '999px',
      background: 'transparent',
      fontSize: '12px',
      lineHeight: '18px',
      whiteSpace: 'nowrap'
    }

    function todayInfo(now) {
      const at = now || new Date()
      const y = at.getFullYear()
      const m = at.getMonth() + 1
      const d = at.getDate()
      const day = QIYAO.dayNumber(y, m, d)
      const planet = DATA[QIYAO.weekdayKey(y, m, d)]
      return {
        y: y,
        m: m,
        d: d,
        planet: planet,
        pick: QIYAO.pickFor(planet, day),
        fit: QIYAO.fitFor(planet, day)
      }
    }

    function QiyaoAction(props) {
      const panelRef = React.useRef(null)
      const info = todayInfo()
      const planet = info.planet
      const sessionId = props && props.sessionId
      const [items, setItems] = React.useState(null)
      const [adding, setAdding] = React.useState(false)
      const [draft, setDraft] = React.useState('')
      const [message, setMessage] = React.useState('')
      const [busy, setBusy] = React.useState(false)
      const [readError, setReadError] = React.useState('')

      function unwrap(res) {
        if (!res) return null
        if (res.ok === false) {
          const err = res.error || {}
          throw new Error(err.message || err.code || '读取失败')
        }
        if (res.text === undefined && res.value !== undefined) return res.value
        return res
      }

      /** 事项由宿主读：客户端只发 /qiyao items，拿回 JSON 再渲染 */
      function fetchItems() {
        const remote = PLUGIN_CTX && PLUGIN_CTX.remote
        if (!remote || !remote.commands) {
          setReadError('浮层拿不到命令通道')
          return Promise.resolve()
        }
        if (!sessionId) {
          setReadError('浮层拿不到会话标识')
          return Promise.resolve()
        }
        return Promise.resolve(remote.commands.execute(sessionId, '/qiyao items', []))
          .then((res) => {
            const data = unwrap(res)
            const text = (data && (data.text || data.value)) || ''
            const parsed = JSON.parse(String(text))
            setReadError('')
            setItems({ today: parsed.today || [], todayTotal: parsed.todayTotal || 0 })
          })
          .catch((error) => {
            setReadError('读不到事项：' + (error && error.message ? error.message : String(error)))
          })
      }

      React.useEffect(() => {
        let alive = true
        fetchItems().catch(() => {
          if (alive) setItems(null)
        })
        return () => {
          alive = false
        }
      }, [sessionId])

      function submitItem() {
        const text = draft.trim()
        if (!text || busy) return
        const remote = PLUGIN_CTX && PLUGIN_CTX.remote
        if (!remote || !remote.commands || !sessionId) {
          setMessage('这个界面读不到宿主的命令通道')
          return
        }
        setBusy(true)
        Promise.resolve(remote.commands.execute(sessionId, '/qiyao add ' + text, []))
          .then((res) => {
            const data = unwrap(res)
            const out = (data && data.text) || (typeof data === 'string' ? data : '')
            setMessage(out ? String(out).split('\n')[0] : '已加上')
            setDraft('')
            setAdding(false)
            return fetchItems()
          })
          .catch((error) => {
            setMessage('没加上：' + (error && error.message ? error.message : String(error)))
          })
          .then(() => setBusy(false))
      }

      function toggle() {
        const panel = panelRef.current
        if (!panel) return
        if (typeof panel.showPopover === 'function') {
          if (panel.matches(':popover-open')) panel.hidePopover()
          else panel.showPopover()
          return
        }
        panel.style.display = panel.style.display === 'block' ? 'none' : 'block'
      }

      const chips = info.fit.map((item) =>
        h(
          'span',
          {
            key: item,
            style: Object.assign({}, chipStyle, {
              border: '1px solid color-mix(in srgb, ' + planet.accent + ' 24%, transparent)'
            })
          },
          item
        )
      )

      return h(
        React.Fragment,
        null,
        h(
          'button',
          {
            type: 'button',
            onClick: toggle,
            title: '七曜日程参考',
            style: {
              display: 'flex',
              alignItems: 'center',
              gap: '4px',
              height: '32px',
              padding: '0 10px',
              borderRadius: '16px',
              border: '1px solid var(--dsw-alias-border-weak, rgba(0,0,0,0.12))',
              background: 'transparent',
              color: 'inherit',
              font: 'inherit',
              fontSize: '13px',
              cursor: 'pointer'
            }
          },
          h('span', { style: { fontSize: '14px', color: planet.accent } }, planet.glyph),
          h('span', null, planet.name)
        ),
        h(
          'div',
          {
            ref: panelRef,
            id: POPOVER_ID,
            popover: 'auto',
            style: {
              position: 'fixed',
              top: '52px',
              right: '20px',
              left: 'auto',
              bottom: 'auto',
              margin: 0,
              width: '360px',
              maxWidth: 'calc(100vw - 40px)',
              padding: '16px 18px',
              borderRadius: '14px',
              border: '1px solid var(--dsw-alias-border-weak, rgba(0,0,0,0.12))',
              background: 'var(--dsw-alias-bg-elevated, #ffffff)',
              boxShadow: '0 12px 36px rgba(0,0,0,0.18)',
              color: 'var(--dsw-alias-label-primary, inherit)',
              fontSize: '13px',
              lineHeight: '1.7'
            }
          },
          h(
            'div',
            { style: { display: 'flex', alignItems: 'baseline', justifyContent: 'space-between' } },
            h(
              'div',
              { style: { fontWeight: 600, fontSize: '15px' } },
              planet.name,
              h(
                'span',
                { style: { fontWeight: 500, fontSize: '13px', opacity: 0.75 } },
                '（',
                h('span', { style: { color: planet.accent } }, planet.glyph),
                ' ' + planet.planet + '）'
              )
            ),
            h(
              'div',
              {
                style: {
                  fontSize: '12px',
                  opacity: 0.42,
                  fontVariantNumeric: 'tabular-nums',
                  whiteSpace: 'nowrap'
                }
              },
              info.m + ' 月 ' + info.d + ' 日'
            )
          ),
          h(
            'div',
            { style: { marginTop: '4px', fontSize: '12px', opacity: 0.66, letterSpacing: '0.02em' } },
            h('span', { style: { color: planet.accent } }, '✦'),
            ' ' + planet.theme
          ),

          h(
            'div',
            {
              style: {
                marginTop: '12px',
                padding: '9px 12px',
                borderRadius: '10px',
                borderLeft: '3px solid ' + planet.accent,
                background: 'color-mix(in srgb, ' + planet.accent + ' 7%, transparent)'
              }
            },
            h(
              'div',
              { style: { fontSize: '11px', opacity: 0.5, marginBottom: '2px', letterSpacing: '0.04em' } },
              '今日首推'
            ),
            h('div', { style: { fontSize: '14px', fontWeight: 600, lineHeight: '1.5' } }, info.pick)
          ),

          h(
            'div',
            { style: { marginTop: '12px', fontSize: '11px', opacity: 0.5, marginBottom: '6px' } },
            '也适合'
          ),
          h(
            'div',
            {
              style: {
                display: 'grid',
                gridTemplateColumns: 'repeat(3, max-content)',
                gap: '7px 6px',
                alignItems: 'center'
              }
            },
            chips
          ),

          items && items.today.length
            ? h(
                'div',
                {
                  style: {
                    marginTop: '12px',
                    paddingTop: '10px',
                    borderTop: '1px solid var(--dsw-alias-border-weak, rgba(0,0,0,0.07))'
                  }
                },
                items.today.length
                  ? h(
                      'div',
                      null,
                      h(
                        'div',
                        { style: { fontSize: '11px', opacity: 0.5, marginBottom: '4px' } },
                        '今天要做'
                      ),
                      items.today.map((item, index) =>
                        h(
                          'div',
                          { key: index, style: { fontSize: '13px', lineHeight: '1.65' } },
                          h(
                            'span',
                            { style: { color: planet.accent, marginRight: '6px' } },
                            '·'
                          ),
                          item
                        )
                      ),
                      items.todayTotal > items.today.length
                        ? h(
                            'div',
                            { style: { fontSize: '11px', opacity: 0.45, marginTop: '2px' } },
                            '还有 ' + (items.todayTotal - items.today.length) + ' 条，在 /qiyao 里'
                          )
                        : null
                    )
                  : null
              )
            : null,
          h(
            'div',
            {
              style: {
                marginTop: '12px',
                paddingTop: '8px',
                borderTop: '1px solid var(--dsw-alias-border-weak, rgba(0,0,0,0.07))',
                fontSize: '11px'
              }
            },
            readError
              ? h(
                  'div',
                  { style: { opacity: 0.5, lineHeight: '1.6', marginBottom: '4px', wordBreak: 'break-all' } },
                  readError
                )
              : null,
            message ? h('div', { style: { opacity: 0.6, lineHeight: '1.6', marginBottom: '4px' } }, message) : null,
            adding
              ? h(
                  'div',
                  { style: { display: 'flex', gap: '6px', alignItems: 'center' } },
                  h('input', {
                    value: draft,
                    autoFocus: true,
                    placeholder: '明天 发作品',
                    onChange: (e) => setDraft(e.target.value),
                    onKeyDown: (e) => {
                      if (e.key === 'Enter') {
                        e.preventDefault()
                        submitItem()
                      }
                      if (e.key === 'Escape') {
                        setAdding(false)
                        setDraft('')
                      }
                    },
                    style: {
                      flex: '1 1 auto',
                      minWidth: 0,
                      height: '26px',
                      padding: '0 8px',
                      borderRadius: '6px',
                      border: '1px solid var(--dsw-alias-border-weak, rgba(0,0,0,0.16))',
                      background: 'transparent',
                      color: 'inherit',
                      font: 'inherit',
                      fontSize: '12px'
                    }
                  }),
                  h(
                    'button',
                    {
                      type: 'button',
                      onClick: submitItem,
                      disabled: busy,
                      style: {
                        height: '26px',
                        padding: '0 10px',
                        borderRadius: '6px',
                        border: 'none',
                        background: 'color-mix(in srgb, ' + planet.accent + ' 16%, transparent)',
                        color: 'inherit',
                        font: 'inherit',
                        fontSize: '12px',
                        cursor: busy ? 'default' : 'pointer'
                      }
                    },
                    busy ? '加…' : '加'
                  ),
                  h(
                    'button',
                    {
                      type: 'button',
                      onClick: () => {
                        setAdding(false)
                        setDraft('')
                      },
                      style: {
                        height: '26px',
                        width: '22px',
                        border: 'none',
                        background: 'transparent',
                        color: 'inherit',
                        opacity: 0.5,
                        font: 'inherit',
                        fontSize: '14px',
                        cursor: 'pointer'
                      }
                    },
                    '×'
                  )
                )
              : h(
                  'button',
                  {
                    type: 'button',
                    onClick: () => {
                      setAdding(true)
                      setMessage('')
                    },
                    style: {
                      all: 'unset',
                      display: 'block',
                      width: '100%',
                      opacity: 0.45,
                      fontSize: '11px',
                      cursor: 'pointer'
                    }
                  },
                  items && items.today.length
                    ? '＋ 再加一件事'
                    : '＋ 加一件事'
                )
          )
        )
      )
    }

    return {
      inject: ['slots', 'remote', 'remote.commands'],
      apply(ctx) {
        PLUGIN_CTX = ctx
        ctx.slots.inject('conversation.session.header.actions', () =>
          ctx.slots.register(
            {
              name: 'conversation.session.header.actions',
              id: 'qiyao-calendar',
              order: 15
            },
            QiyaoAction
          )
        )
      }
    }
  }
})
